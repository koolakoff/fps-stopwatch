cmake_minimum_required(VERSION 3.20)
project(FpsStopwatch VERSION 1.0.0 LANGUAGES CXX RC)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(fps-stopwatch WIN32 src/main.cpp src/app.rc)
target_compile_features(fps-stopwatch PRIVATE cxx_std_17)
target_compile_definitions(fps-stopwatch PRIVATE UNICODE _UNICODE WINVER=0x0A00 _WIN32_WINNT=0x0A00 NOMINMAX)
target_link_libraries(fps-stopwatch PRIVATE d3d11 dxgi d2d1 dwrite user32 gdi32 ole32 windowscodecs)
if(MSVC)
  set_property(TARGET fps-stopwatch PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_compile_options(fps-stopwatch PRIVATE /W4 /utf-8 /EHsc)
  target_link_options(fps-stopwatch PRIVATE /MANIFEST:NO)
else()
  target_compile_options(fps-stopwatch PRIVATE -Wall -Wextra)
  target_link_options(fps-stopwatch PRIVATE -municode -static)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(stopwatch-test tests/stopwatch_test.cpp)
  target_compile_features(stopwatch-test PRIVATE cxx_std_17)
  if(MSVC)
    target_compile_options(stopwatch-test PRIVATE /W4 /utf-8)
  else()
    target_compile_options(stopwatch-test PRIVATE -UNDEBUG)
  endif()
  add_test(NAME stopwatch COMMAND stopwatch-test)
endif()
